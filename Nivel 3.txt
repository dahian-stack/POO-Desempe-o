from abc import ABC, abstractmethod


class Habitacion:
    def __init__(self, numero, tipo, precio_noche):
        self.numero = numero
        self.tipo = tipo
        self.precio_noche = precio_noche

    def obtener_precio(self):
        return self.precio_noche

    def obtener_info(self):
        return f"Habitación {self.numero} - Tipo: {self.tipo} - Precio por noche: ${self.precio_noche}"


class Reserva(ABC):
    def __init__(self, cliente, habitacion, noches):
        self.cliente = cliente
        self.habitacion = habitacion
        self.noches = noches

    @abstractmethod
    def calcular_total(self):
        pass

    @abstractmethod
    def confirmar(self):
        pass


class ReservaEstandar(Reserva):
    def __init__(self, cliente, habitacion, noches):
        super().__init__(cliente, habitacion, noches)

    def calcular_total(self):
        precio = self.habitacion.obtener_precio()
        return precio * self.noches

    def confirmar(self):
        total = self.calcular_total()
        return (f"Reserva estandar confirmada\n"
                f"Cliente: {self.cliente}\n"
                f"{self.habitacion.obtener_info()}\n"
                f"Noches: {self.noches}\n"
                f"Total a pagar: $ {total}")


class ReservaVIP(Reserva):
    def __init__(self, cliente, habitacion, noches):
        super().__init__(cliente, habitacion, noches)
        self.valor_desayuno = 25000

    def calcular_total(self):
        precio_base = self.habitacion.obtener_precio()
        return (precio_base + self.valor_desayuno) * self.noches

    def confirmar(self):
        total = self.calcular_total()
        return (f"Reserva VIP confirmada (Incluye desayuno)\n"
                f"Cliente: {self.cliente}\n"
                f"{self.habitacion.obtener_info()}\n"
                f"Noches: {self.noches}\n"
                f"Valor desayuno/noche: $ {self.valor_desayuno}\n"
                f"Total a pagar: $ {total}")


if __name__ == "__main__":
    hab1 = Habitacion(101, "Doble", 100000)

    reserva1 = ReservaEstandar("Esteban Sanchez", hab1, 3)
    reserva2 = ReservaVIP("Elizabeth Perez", hab1, 3)

    print(reserva1.confirmar())
